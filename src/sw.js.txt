// Service Worker der Health-Hub-App: nur für Push-Meldungen (#13), kein Offline-Zwischenspeicher.
self.addEventListener('install', () => self.skipWaiting());
self.addEventListener('activate', (e) => e.waitUntil(self.clients.claim()));
self.addEventListener('push', (e) => {
  let m = {};
  try { m = e.data ? e.data.json() : {}; } catch { m = { title: 'Health Hub', body: e.data ? e.data.text() : '' }; }
  e.waitUntil(self.registration.showNotification(m.title || 'Health Hub', {
    body: m.body || '', tag: m.tag || 'health-hub', icon: '/icon-192.png', badge: '/icon-192.png', data: { url: m.url || '/' },
  }));
});
// Tippen auf die Meldung öffnet die App auf „Heute“ (bzw. holt das offene Fenster nach vorn)
self.addEventListener('notificationclick', (e) => {
  e.notification.close();
  const url = new URL((e.notification.data && e.notification.data.url) || '/', self.location.origin).href;
  e.waitUntil(self.clients.matchAll({ type: 'window', includeUncontrolled: true }).then((list) => {
    for (const c of list) if ('focus' in c) { c.navigate(url).catch(() => {}); return c.focus(); }
    return self.clients.openWindow(url);
  }));
});
