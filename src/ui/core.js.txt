// Kalenderwoche nach ISO 8601
function isoWeek(d) {
  const t = parse(addDays(d, 3 - (parse(d).getUTCDay() + 6) % 7));
  const jan4 = parse(t.getUTCFullYear() + '-01-04');
  return 1 + Math.round(((t - jan4) / 864e5 - 3 + (jan4.getUTCDay() + 6) % 7) / 7);
}

// Zeitraum und Vergleichszeitraum. „1 W“ ist die Kalenderwoche Mo–So (S.weekOffset 0 = aktuelle,
// -1 = letzte …). Läuft die Woche noch, endet sie heute, und verglichen wird mit demselben Stück
// der Vorwoche (Mi gegen Mi). Abgeschlossene Wochen werden komplett mit der Woche davor verglichen.
function ranges() {
  const end = S.today;
  if (S.days !== 7) return { cur: [addDays(end, -S.days + 1), end], prev: [addDays(end, -2 * S.days + 1), addDays(end, -S.days)] };
  const mon = addDays(weekStart(end), 7 * S.weekOffset), sun = addDays(mon, 6);
  const last = sun < end ? sun : end;
  return { cur: [mon, last], prev: [addDays(mon, -7), addDays(last, -7)], running: last < sun };
}

// Beschriftung des Vergleichs, z. B. „ggü. Vorwoche (Mo–Mi)“
function cmpName() {
  if (S.days !== 7) return 'ggü. davor';
  const r = ranges();
  return r.running ? 'ggü. Vorwoche (Mo–' + WD[parse(r.cur[1]).getUTCDay()] + ')' : 'ggü. Vorwoche';
}

// Vergleichszeile: Wert gegen Vorperiode, Farbe je nachdem ob höher besser ist
function cmpLine(cur, prev, digits, betterUp, unit = '') {
  if (cur == null || prev == null) return { text: 'kein Vergleich', cls: '' };
  const d = cur - prev;
  const r = Number(d.toFixed(digits));
  const sign = r > 0 ? '+' : r < 0 ? '−' : '±';
  const cls = r === 0 || betterUp == null ? '' : (r > 0) === betterUp ? 'good' : 'bad';
  return { text: sign + num(Math.abs(d), digits) + unit + ' ' + cmpName(), cls };
}

function frame(node, h) {
  node.replaceChildren();
  // Im Vollbild (#61) deutlich höher, aber nie höher als gut die Hälfte des Bildschirms
  if (node.closest('#dg-body')) h = Math.round(Math.max(h, Math.min(h * 1.8, window.innerHeight * 0.55)));
  const w = Math.max(260, Math.round(node.clientWidth || 320));

  const tip = d3.select(node).append('div').attr('class', 'tip');
  return { svg, w, h, tip, node };
}

function xTime(f, x, yPos) {
  const [a, b] = x.domain(); const days = (b - a) / 864e5;

  else if (days <= 35) { ticks = x.ticks(d3.utcMonday.every(1)); fmt = (t) => t.getUTCDate() + '.' + (t.getUTCMonth() + 1) + '.'; }
  else if (days <= 130) { ticks = x.ticks(d3.utcMonday.every(f.w < 420 ? 3 : 2)); fmt = (t) => t.getUTCDate() + '.' + (t.getUTCMonth() + 1) + '.'; }
  else { ticks = x.ticks(d3.utcMonth.every(f.w < 420 && days > 200 ? 2 : 1)); fmt = (t) => MONTHS[t.getUTCMonth()]; }

  }
  const s = f.node.clientWidth / f.w, n = f.tip.node();
  let left = px * s + 12; if (left + n.offsetWidth > f.node.clientWidth) left = px * s - n.offsetWidth - 12;
  f.tip.style('left', Math.max(0, left) + 'px').style('top', Math.max(0, py * s - n.offsetHeight - 10) + 'px').style('opacity', 1);

function hoverLayer(f, box, find, withRule = true) {
  const g = f.svg.append('g').style('display', 'none').style('pointer-events', 'none');
  const rule = withRule ? g.append('line').attr('y1', box.y0).attr('y2', box.y1).style('stroke', css('--muted')).style('stroke-dasharray', '3 3') : null;
  const dot = g.append('circle').attr('r', 4.5).style('stroke', css('--panel')).style('stroke-width', 2);
  const move = (ev) => {
    const [px, py] = d3.pointer(ev, f.svg.node());
    const p = find(px, py); if (!p) return;
    g.style('display', null);
    if (rule) rule.attr('x1', p.x).attr('x2', p.x);
    dot.attr('cx', p.x).attr('cy', p.y).style('fill', p.color || css('--accent'));
    showTip(f, p.x, p.y, p.title, p.lines);
  };
  f.svg.append('rect').attr('x', box.x0).attr('y', box.y0).attr('width', box.x1 - box.x0).attr('height', box.y1 - box.y0)
    .style('fill', 'transparent').style('pointer-events', 'all')
    .on('pointermove', move).on('pointerdown', move)
    .on('pointerleave', () => { g.style('display', 'none'); f.tip.style('opacity', 0); });
}
function fit(points) {
  const n = points.length; if (n < 4) return null;
  const mx = d3.mean(points, (p) => p[0]), my = d3.mean(points, (p) => p[1]);
  let sxy = 0, sxx = 0; for (const [x, y] of points) { sxy += (x - mx) * (y - my); sxx += (x - mx) ** 2; }
  if (!sxx) return null; const b = sxy / sxx; return { f: (x) => my + b * (x - mx), b };
}
// Linie mit Verlauf darunter: 30 % Deckkraft direkt unter der Linie, nach unten auf 0 %.
// Weich gezeichnet (monotone Kurve, keine Überschwinger). opts: x, y (Funktionen je Punkt), base (y-Pixel der Grundlinie),
// color, width, defined, area (false = nur Linie), dot (letzten Punkt hervorheben)
let gradSeq = 0;
function glowLine(f, data, { x, y, base, color = css('--accent'), width = 2.5, defined = () => true, area = true, dot = false, top = .3 }) {
  const pts = data.filter(defined);
  if (!pts.length) return;
  const curve = d3.curveMonotoneX;
  if (area && base != null) {
    const id = 'gl' + (++gradSeq);
    const g = f.svg.append('defs').append('linearGradient').attr('id', id).attr('x1', 0).attr('x2', 0).attr('y1', 0).attr('y2', 1);
    g.append('stop').attr('offset', '0%').attr('stop-color', color).attr('stop-opacity', top);
    g.append('stop').attr('offset', '100%').attr('stop-color', color).attr('stop-opacity', 0);
    f.svg.append('path').datum(data).attr('fill', `url(#${id})`)
      .attr('d', d3.area().defined(defined).curve(curve).x(x).y0(base).y1(y));
  }
  f.svg.append('path').datum(data).style('fill', 'none').style('stroke', color).style('stroke-width', width)
    .style('stroke-linejoin', 'round').style('stroke-linecap', 'round')
    .attr('d', d3.line().defined(defined).curve(curve).x(x).y(y));
  if (dot) {
    const last = pts[pts.length - 1];
    f.svg.append('circle').attr('cx', x(last)).attr('cy', y(last)).attr('r', 7).style('fill', color).style('opacity', .25);
    f.svg.append('circle').attr('cx', x(last)).attr('cy', y(last)).attr('r', 3.5).style('fill', color).style('stroke', css('--panel')).style('stroke-width', 1.5);
  }
}
function sparkline(node, data, band) {
  const v = data.filter((d) => d.v != null);
  if (v.length < 2) return;
  const f = frame(node, 42);
  const vals = v.map((d) => d.v).concat(band ? band.flatMap((b) => [b.lo, b.hi]).filter((x) => x != null) : []);
  const y = d3.scaleLinear(d3.extent(vals), [34, 7]);
  // rechts Platz für den hervorgehobenen letzten Punkt
  const x = d3.scaleUtc(d3.extent(data, (d) => parse(d.d)), [2, f.w - 9]);
  if (band) f.svg.append('path').datum(band.filter((b) => b.lo != null)).style('fill', css('--prev')).style('opacity', .25)
    .attr('d', d3.area().x((b) => x(parse(b.d))).y0((b) => y(b.lo)).y1((b) => y(b.hi)));
  glowLine(f, v, { x: (d) => x(parse(d.d)), y: (d) => y(d.v), base: f.h, width: 2, dot: true });
}

// ---------- Berechnungen ----------
function dailyLoad() {
  const m = new Map();
  for (const a of S.acts) m.set(a.date, (m.get(a.date) || 0) + (a.load || 0));
  return m;
}
function loadSeries(from, to) {
  const m = dailyLoad(), out = [];
  for (let d = addDays(from, -27); d <= to; d = addDays(d, 1)) out.push({ d, v: m.get(d) || 0 });
  return out.map((r, i) => ({
    ...r,
    acute: i >= 6 ? mean(out.slice(i - 6, i + 1).map((x) => x.v)) : null,
    chronic: i >= 27 ? mean(out.slice(i - 27, i + 1).map((x) => x.v)) : null
  })).filter((r) => r.d >= from);
}
// Balken-Abschnitte: bei 1 Woche einzelne Tage, sonst Wochen ab Montag
const bucketDays = () => (S.days <= 7 ? 1 : 7);
function weeksIn(r) {
  if (bucketDays() === 1) { const days = []; for (let d = r[0]; d <= r[1]; d = addDays(d, 1)) days.push(d); return days; }
  const weeks = []; let d = r[0];
  const dow = (parse(d).getUTCDay() + 6) % 7; d = addDays(d, -dow);
  for (; d <= r[1]; d = addDays(d, 7)) weeks.push(d);
  return weeks;
}
