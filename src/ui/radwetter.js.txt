// ---------- RAD-WETTER (#153): Wann kann ich fahren, was ziehe ich an? ----------
// Eigene Unterseite (#radwetter/<ort>), dazu eine Zeile auf „Heute“, eine Karte unter „Einheiten“ und die Rückmeldung
// „Wie war's angezogen?“ auf der Seite einer Radfahrt. Der Server rechnet alles (src/radwetter.ts), hier wird nur angezeigt.
// Einstellungen unter Konto → Rad-Wetter (radwetter-konto.js.txt).
const RW = { data: null, key: '', loading: false, hour: null, dur: null, here: null, pick: null, fig: null, short: null, shortAt: 0, tick: null, err: '' };
const RW_DUR = [[60, '1 h'], [120, '2 h'], [180, '3 h'], [240, '4 h']];
const capFirst = (s) => (s ? s.charAt(0).toUpperCase() + s.slice(1) : '');
/** Dauer als „42 Min.“, „2:14 h“ oder „1 Tag 3 h“ */
function rwDur(sec) {
  const m = Math.max(0, Math.round(sec / 60));
  if (m < 60) return m + ' Min.';
  if (m < 1440) return Math.floor(m / 60) + ':' + String(m % 60).padStart(2, '0') + ' h';
  const d = Math.floor(m / 1440), h = Math.round((m % 1440) / 60);
  return d + (d === 1 ? ' Tag' : ' Tage') + (h ? ' ' + h + ' h' : '');
}
/** Countdown zum Fenster: vor dem Start „In …“, währenddessen „Jetzt“ mit Restzeit */
function rwCd(w) {
  const now = Date.now() / 1000;
  if (!w) return null;
  if (w.s <= now && w.e > now) return { live: true, big: 'Jetzt', sub: 'noch ' + rwDur(w.e - now) + ' gut' };
  if (w.e <= now) return { live: false, big: 'Vorbei', sub: '' };
  return { live: false, big: 'In ' + rwDur(w.s - now), sub: '' };
}
/** Alle Countdowns auf der Seite (Unterseite, Heute, Einheiten) alle 20 Sekunden nachführen */
function rwTickStart() {
  if (RW.tick) return;
  RW.tick = setInterval(() => {
    if (document.visibilityState !== 'visible') return;
    document.querySelectorAll('[data-rw-cd]').forEach((n) => {
      const [s, e] = n.dataset.rwCd.split(',').map(Number), c = rwCd({ s, e });
      if (!c) return;
      const big = n.querySelector('.rw-cd-big'), sub = n.querySelector('.rw-cd-sub');
      if (big) big.textContent = c.big; if (sub) sub.textContent = c.sub;
      n.classList.toggle('live', c.live);
    });
  }, 20000);
}
const rwWxIcon = (h) => { const [ic, label] = wxKind(h.c); return [h.light < 30 && h.c <= 2 ? 'mond' : ic, label]; };
const rwPost = (body) => api('/api/radwetter', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });

function openRadwetter(place) { openSub('radwetter', place || 'start'); }

async function loadRw(force) {
  if (RW.loading) return;
  const key = (S.sub && S.sub.key) || 'start';
  const q = new URLSearchParams();
  if (key !== 'start' && key !== 'hier') q.set('place', key);
  if (key === 'hier' && RW.here) { q.set('lat', RW.here[0]); q.set('lng', RW.here[1]); }
  if (RW.dur) q.set('dur', RW.dur);
  const want = q.toString();
  if (!force && RW.data && RW.key === want) return;
  RW.loading = true;
  try { RW.data = await api('/api/radwetter?' + want); RW.key = want; RW.err = ''; }
  catch (e) { RW.err = e.message || 'Fehler'; }
  RW.loading = false;
  RW.short = null; // Kurzfassung für Heute/Einheiten neu holen
  if (S.tab === 'radwetter') renderRadwetter(true);
}

function renderRadwetter(fresh) {
  const body = $('#rw-body');
  const key = (S.sub && S.sub.key) || 'start';
  $('#navTitle').textContent = 'Rad-Wetter';
  rwTickStart();
  const D = RW.data;
  const want = key === 'hier' ? 'hier' : key;
  // Andere Seite gewünscht als geladen: neu laden
  if (!D || (want !== 'start' && D.sel && D.sel !== want && !(want === 'hier' && !RW.here))) {
    if (!RW.loading) { body.replaceChildren(el('div', 'card rw-empty', RW.err ? 'Konnte nicht laden: ' + RW.err : 'Lade Vorhersage …')); loadRw(); }
    return;
  }
  if (!fresh && body.dataset.key === RW.key + '|' + (RW.hour ?? '')) return;
  body.dataset.key = RW.key + '|' + (RW.hour ?? '');
  body.replaceChildren();
  $('#rw-sub').textContent = D.on && D.at ? 'Stand ' + new Date(D.at).toLocaleTimeString('de-DE', { hour: '2-digit', minute: '2-digit' }) + ' Uhr' + (D.place ? ' · ' + D.place.name : '') : 'Wann kann ich fahren?';
  if (!D.on || !D.places.length) { body.append(rwIntro(D)); return; }
  body.append(rwPlaces(D));
  if (D.err) body.append(el('div', 'banner err', 'Die Vorhersage konnte nicht aktualisiert werden: ' + D.err + (D.at ? ' Angezeigt wird der letzte Stand.' : '')));
  if (!D.hours || !D.hours.length) { body.append(el('div', 'card rw-empty', 'Für diesen Ort gibt es noch keine Vorhersage. Sie kommt beim nächsten Abruf.')); body.append(rwFoot(D)); return; }
  if (RW.hour == null || RW.hour >= D.hours.length) RW.hour = D.wearStart || 0;
  body.append(rwHero(D), rwTimeline(D), rwWearCard(D), rwWindCard(D));
  const fb = rwFeedbackCard(D); if (fb) body.append(fb);
  if (D.game) body.append(...rwGame(D));
  body.append(rwFoot(D));
  rwSelectHour(RW.hour, false);
}

async function rwSetHome(body, msg) {
  msg.textContent = 'Speichere …';
  try { const r = await rwPost({ action: 'home', ...body }); if (!r.ok) { msg.textContent = r.error || 'Fehler'; return; } msg.textContent = ''; RW.data = null; RW.short = null; loadRw(true); }
  catch (e) { msg.textContent = 'Fehler: ' + e.message; }
}
/** Standort nur nach Antippen, auf 0,01° (etwa 1 km) gerundet */
function rwLocate(msg, done) {
  if (!navigator.geolocation) { msg.textContent = 'Dieses Gerät kann den Standort nicht bestimmen.'; return; }
  msg.textContent = 'Bestimme Standort …';
  navigator.geolocation.getCurrentPosition((p) => { msg.textContent = ''; done(Math.round(p.coords.latitude * 100) / 100, Math.round(p.coords.longitude * 100) / 100); },
    (e) => { msg.textContent = e.code === 1 ? 'Ohne Erlaubnis kein Standort. Du kannst stattdessen einen Ort suchen.' : 'Standort nicht verfügbar.'; },
    { enableHighAccuracy: false, timeout: 15000, maximumAge: 600000 });
}

// ---------- Orte oben ----------
function rwPlaces(D) {
  const box = el('div', 'rw-places'); box.setAttribute('role', 'group'); box.setAttribute('aria-label', 'Ort');
  const add = (id, name, ic, next) => {
    const b = el('button', 'rw-place'); b.type = 'button';
    b.setAttribute('aria-pressed', String(D.sel === id));
    if (ic) b.innerHTML = icon(ic);
    b.append(el('span', null, name));
    if (next) { const d = el('i', 'rw-dot'); d.title = 'Gutes Fenster ' + next.day + ' ' + next.time; b.append(d); }
    b.addEventListener('click', () => {
      if (id === 'hier') { const m = $('#rw-here-msg'); rwLocate(m, (lat, lng) => { RW.here = [lat, lng]; rwGo('hier'); }); return; }
      rwGo(id);
    });
    box.append(b);
  };
  for (const p of D.places) add(p.id, p.name, p.kind === 'home' ? 'haus' : 'route', p.next);
  add('hier', 'Hier', 'standort', null);
  const wrap = el('div', 'rw-places-wrap'); wrap.append(box);
  const m = el('p', 'msg rw-here-msg'); m.id = 'rw-here-msg'; m.setAttribute('role', 'status');
  if (D.sel === 'hier') m.textContent = 'Dein Standort, auf etwa 1 km gerundet und nicht gespeichert.';
  wrap.append(m);
  return wrap;
}
function rwGo(id) {
  RW.hour = null; RW.pick = null;
  if (S.sub) S.sub.key = id;
  history.replaceState(history.state, '', '#radwetter/' + encodeURIComponent(id));
  loadRw();
  renderRadwetter(true);
}

  const c = el('div', 'card rw-hero');
  const w = D.next;
  const top = el('div', 'rw-hero-top');
  top.append(el('span', 'rw-lbl', w ? (rwCd(w).live ? 'Gutes Fenster läuft' : 'Nächstes gutes Fenster') : 'Kein gutes Fenster in Sicht'));
  if (w) { const q = el('span', 'rw-q', quality(w.score)); top.append(q); }
  c.append(top);
  if (w) {
    const cd = rwCd(w), box = el('div', 'rw-cd' + (cd.live ? ' live' : ''));
    box.dataset.rwCd = w.s + ',' + w.e;
    box.append(el('span', 'rw-cd-big', cd.big), el('span', 'rw-cd-sub', cd.sub));
    c.append(box);
    c.append(el('p', 'rw-when', capFirst(w.day) + ' ' + w.time));
    c.append(el('p', 'rw-facts', rwWinFacts(w)));
    c.addEventListener('click', (e) => { if (!e.target.closest('button')) rwSelectHour(Math.max(0, w.k), true); });
  } else {
    c.append(el('p', 'rw-when', 'In den nächsten 48 Stunden passt nichts zu deinen Kriterien.'));
    if (D.near) c.append(el('p', 'rw-facts', 'Am nächsten dran: ' + D.near.day + ' ' + D.near.time + (D.near.why.length ? ' (' + D.near.why.slice(0, 3).join(', ') + ')' : '') + '.'));
  }
  if (D.best && (!w || D.best.s !== w.s)) {
    const b = el('button', 'rw-best'); b.type = 'button';
    b.append(el('span', null, 'Bestes Fenster: ' + capFirst(D.best.day) + ' ' + D.best.time), el('span', 'rw-q', quality(D.best.score)));
    const ch = el('span', 'st-chev'); ch.innerHTML = icon('weiter'); b.append(ch);
    b.addEventListener('click', () => rwSelectHour(Math.max(0, D.best.k), true));
    c.append(b);
  }
  // Andere Orte mit Fenster
  const others = D.places.filter((p) => p.id !== D.sel && p.next);
  if (others.length) c.append(el('p', 'note rw-others', 'Auch gut: ' + others.map((p) => p.name + ' ' + p.next.day + ' ' + p.next.time).join(' · ')));
  return c;
}
const rwRange = (a, b, u) => (Math.round(a) === Math.round(b) ? num(a) : num(a) + '–' + num(b)) + u;
function rwWinFacts(w) {
  const t = rwRange(w.t[0], w.t[1], '');
  return t + ' °C · Wind ' + w.ws + ' km/h aus ' + windDir(w.wdir) + ' · trocken' + (w.sun >= 60 ? ' · ' + hm(w.sun / 60) + ' h Sonne' : '');
}
const quality = (s) => (s >= 80 ? 'sehr gut' : s >= 60 ? 'gut' : 'okay');

function rwTimeline(D) {
  const c = el('div', 'card rw-tl');
  const head = el('div', 'rw-card-head'); head.append(el('h3', null, '48 Stunden'));
  const leg = el('div', 'rw-legend');
  for (const [cls, t] of [['good', 'gut'], ['meh', 'knapp'], ['off', 'nicht deine Zeit']]) { const s = el('span'); s.append(el('i', 'rw-sw is-' + cls), document.createTextNode(t)); leg.append(s); }
  head.append(leg); c.append(head);
  // Überblick: ein Streifen je Stunde, Fenster darüber, Tage darunter
  const band = el('div', 'rw-band'); band.setAttribute('aria-hidden', 'true');
  const n = D.hours.length;
  band.style.setProperty('--n', n);
  for (const w of D.wins) {
    const a = Math.max(0, w.k), b = Math.min(n, w.k + w.n);
    if (b <= a) continue;
    const s = el('span', 'rw-band-win'); s.style.gridColumn = (a + 1) + ' / span ' + (b - a);
    band.append(s);
  }
  D.hours.forEach((h, i) => {
    const b = el('i', 'rw-band-h is-' + (h.off ? 'off' : h.st)); b.style.gridColumn = String(i + 1); b.dataset.i = i;
    band.append(b);
    if (i === 0 || h.h === 0) { const d = el('span', 'rw-band-day', i === 0 ? 'Jetzt' : WD[parse(h.d).getUTCDay()]); d.style.gridColumn = (i + 1) + ' / span 6'; band.append(d); }
  });
  band.addEventListener('click', (e) => {
    const r = band.getBoundingClientRect(), i = Math.floor(((e.clientX - r.left) / r.width) * n);
    if (i >= 0 && i < n) rwSelectHour(i, true);
  });
  c.append(band);
  // Stunden zum Wischen
  const strip = el('div', 'rw-strip'); strip.id = 'rw-strip';
  strip.setAttribute('role', 'listbox'); strip.setAttribute('aria-label', 'Stunden der nächsten 48 Stunden');
  const inWin = new Set(); for (const w of D.wins) for (let k = 0; k < w.n; k++) inWin.add(w.k + k);
  D.hours.forEach((h, i) => {
    const b = el('button', 'rw-h is-' + (h.off ? 'off' : h.st) + (inWin.has(i) ? ' in-win' : '')); b.type = 'button'; b.dataset.i = i;
    b.setAttribute('role', 'option');
    const [ic, label] = rwWxIcon(h);

    b.append(el('span', 'rw-hd', day), el('span', 'rw-hh', String(h.h).padStart(2, '0')));

    const rain = el('span', 'rw-hr'), bar = el('i'); bar.style.height = Math.min(100, (h.p / 2) * 100) + '%'; rain.append(bar); b.append(rain);
    b.append(el('span', 'rw-hp', h.pp >= 10 ? h.pp + ' %' : ''));
    const wind = el('span', 'rw-hw'); wind.innerHTML = icon('pfeil'); wind.style.setProperty('--rot', (h.wdir + 180) + 'deg'); b.append(wind);
    b.append(el('span', 'rw-hws' + (h.ws > ((D.cfg && D.cfg.crit.wind) || 25) ? ' hi' : ''), String(h.ws)));
    b.append(el('span', 'rw-hs'));
    b.setAttribute('aria-label', rwHourLabel(D, h, i) + ', ' + label);
    b.addEventListener('click', () => rwSelectHour(i, false));
    strip.append(b);
  });
  c.append(strip);
  const det = el('div', 'rw-detail'); det.id = 'rw-detail'; det.setAttribute('aria-live', 'polite');
  c.append(det);
  // Wie rechnet die App?
  const more = el('details', 'fold rw-rules');
  more.append(el('summary', null, 'So bewertet die App'));
  const cr = D.cfg.crit;
  const ul = el('ul');
  const li = (t) => ul.append(el('li', null, t));
  li('Gut: Regenrisiko höchstens ' + cr.prob + ' %, Niederschlag ' + (cr.mm ? 'höchstens ' + num(cr.mm, 1) + ' mm' : 'kein messbarer') + ', Wind höchstens ' + cr.wind + ' km/h, Böen höchstens ' + cr.gust + ' km/h' + (cr.tMin != null ? ', ab ' + cr.tMin + ' °C' : '') + (cr.tMax != null ? ', bis ' + cr.tMax + ' °C' : '') + '.');
  li('Ein Fenster dauert mindestens ' + cr.minH + (cr.minH === 1 ? ' Stunde' : ' Stunden') + (cr.light ? ', nur bei Tageslicht' : '') + ', in deinen Zeiten (Mo–Fr ' + cr.wk[0] + '–' + cr.wk[1] + ' Uhr, Sa/So ' + cr.we[0] + '–' + cr.we[1] + ' Uhr).');
  if (cr.dry) li(D.dryRule);
  li('Knapp heißt: nur Wind, Böen, Regenrisiko oder Temperatur ein wenig über deiner Grenze, oder die Straße ist noch feucht.');
  more.append(ul);
  const r = el('div', 'row'), go = el('button', 'btn ghost small', 'Kriterien ändern'); go.type = 'button';
  go.addEventListener('click', () => rwSettings('rwk-crit'));
  r.append(go); more.append(r);
  c.append(more);

function rwHourLabel(D, h, i) {
  const day = h.d === D.today ? 'heute' : dayLabel(h.d).split(',')[0];
  return (i === 0 ? 'Jetzt, ' : '') + day + ' ' + h.h + ' Uhr: ' + num(h.t) + ' °C, ' + (h.p >= 0.1 ? num(h.p, 1) + ' mm Regen' : h.pp >= 10 ? h.pp + ' % Regenrisiko' : 'trocken') + ', Wind ' + h.ws + ' km/h aus ' + windDir(h.wdir)
    + ', ' + (h.off === 'zeit' ? 'nicht in deinen Zeiten' : h.off === 'dunkel' ? 'dunkel' : h.st === 'good' ? 'gut' : h.st === 'meh' ? 'knapp' : 'nicht gut');
}
/** Stunde wählen: Zeitleiste markieren, Details, Kleidung für einen Start in dieser Stunde */
function rwSelectHour(i, scroll) {
  const D = RW.data; if (!D || !D.hours) return;
  RW.hour = i;
  const body = $('#rw-body'); body.dataset.key = RW.key + '|' + i;
  document.querySelectorAll('#rw-strip .rw-h').forEach((b) => { const on = Number(b.dataset.i) === i; b.classList.toggle('sel', on); b.setAttribute('aria-selected', String(on)); });
  document.querySelectorAll('.rw-band-h').forEach((b) => b.classList.toggle('sel', Number(b.dataset.i) === i));
  const strip = $('#rw-strip'), cell = strip && strip.querySelector('[data-i="' + i + '"]');
  if (cell && scroll !== null) {
    const left = cell.offsetLeft - strip.clientWidth / 2 + cell.offsetWidth / 2;
    strip.scrollTo({ left, behavior: scroll && !matchMedia('(prefers-reduced-motion: reduce)').matches ? 'smooth' : 'auto' });
  }
  const h = D.hours[i], det = $('#rw-detail');
  if (det && h) {
    det.replaceChildren();
    const top = el('div', 'rw-detail-top');
    top.append(el('b', null, (i === 0 ? 'Jetzt · ' : '') + capFirst(h.d === D.today ? 'heute' : dayLabel(h.d)) + ', ' + h.h + ' Uhr'));
    const st = h.off === 'zeit' ? ['off', 'Nicht deine Zeit'] : h.off === 'dunkel' ? ['off', 'Dunkel'] : h.st === 'good' ? ['good', 'Gut'] : h.st === 'meh' ? ['meh', 'Knapp'] : ['bad', 'Nicht gut'];
    top.append(el('span', 'rw-st is-' + st[0], st[1]));
    det.append(top);
    const [, label] = rwWxIcon(h);
    det.append(el('p', null, [label, num(h.t) + ' °C, gefühlt ' + num(h.ta) + ' °C, auf dem Rad ' + num(h.f) + ' °C', h.p >= 0.1 ? num(h.p, 1) + ' mm Regen' : h.pp >= 10 ? h.pp + ' % Regenrisiko' : 'trocken',
      'Wind ' + h.ws + ' km/h aus ' + windDir(h.wdir) + ', Böen ' + h.wg, 'Straße ' + h.road].filter(Boolean).join(' · ')));
    if (h.why.length) det.append(el('p', 'note', (h.st === 'meh' ? 'Knapp: ' : 'Passt nicht: ') + h.why.join(', ') + '.'));
  }
  rwRenderWear(i);
}

function rwRenderWear(i) {

/** Skizze einer Runde (Form ohne Karte): Start, Halbzeit, Wind-Pfeil */
function rwSketch(r, W, size) {
  const box = el('span', 'rw-sketch ' + size); box.setAttribute('aria-hidden', 'true');
  if (!r.sk) return box;
  const svg = svgEl('svg', { viewBox: '0 0 100 100' }, null);
  svgEl('polyline', { points: r.sk.map((p) => p.join(',')).join(' '), class: 'rw-sk-line' }, svg);
  const s = r.sk[0], h = r.sk[r.hi ?? Math.floor(r.sk.length / 2)];
  svgEl('circle', { cx: h[0], cy: h[1], r: size === 'big' ? 4 : 6, class: 'rw-sk-half' }, svg);
  svgEl('circle', { cx: s[0], cy: s[1], r: size === 'big' ? 4.5 : 7, class: 'rw-sk-start' }, svg);
  box.append(svg);
  const a = el('span', 'rw-sk-wind'); a.innerHTML = icon('pfeil'); a.style.setProperty('--rot', (W.dir + 180) + 'deg'); box.append(a);
  if (size === 'big') {
    // Legende ohne Zeichen-Symbole: gezeichnete Punkte und das Pfeil-Symbol
    const l = el('span', 'rw-sk-leg');
    const item = (cls, text) => { const s = el('span'); s.append(el('i', cls), document.createTextNode(text)); l.append(s); };
    item('rw-lg start', 'Start'); item('rw-lg half', 'Halbzeit');
    const w = el('span'); w.innerHTML = icon('pfeil'); w.append('Wind'); l.append(w);
    box.append(l);
  }
  return box;
}

function rwFbButtons(id, done) {
  const row = el('div', 'rw-fb-btns');
  for (const [v, t] of [[-1, 'Zu kalt'], [0, 'Passt'], [1, 'Zu warm']]) {
    const b = el('button', 'btn ghost', t); b.type = 'button';
    b.addEventListener('click', async () => {
      row.querySelectorAll('button').forEach((x) => { x.disabled = true; });
      try {
        const r = await rwPost({ action: 'fb', id, v });
        if (!r.ok) throw new Error(r.error || 'Fehler');
        RW.data = null; RW.short = null; RWK.cfg = null;
        done(v === 0 ? 'Gemerkt. Deine Grenzen bleiben, wie sie sind.' : 'Gemerkt. Die App rechnet für dich ab jetzt 0,5 °C ' + (v < 0 ? 'kälter' : 'wärmer') + ' (nachzulesen unter Konto → Rad-Wetter).');
        if (S.tab === 'radwetter') setTimeout(() => loadRw(true), 1200);
      } catch (e) { row.querySelectorAll('button').forEach((x) => { x.disabled = false; }); done('Fehler: ' + e.message); }
    });
    row.append(b);
  }
  return row;
}
