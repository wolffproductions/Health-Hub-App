// ---------- Level und Punkte (#192) ----------
// Rechnung im Worker (src/level.ts), /api/data liefert `level` fertig. Hier nur Anzeige:
// Konto (Karte unter dem Profil, Unterseite „Level“ mit den letzten Punkten und den Regeln) und eine Zeile auf „Heute“.
const lvNum = (n) => Math.round(n).toLocaleString('de-DE');
function lvLine(l) { return 'noch ' + lvNum(l.need - l.into) + ' bis Level ' + (l.level + 1); }

function renderLevel() {
  const l = S.level, card = $('#lv-card');
  if (!card) return;
  card.hidden = !l;
  if (!l) return;
  $('#lv-lvl').textContent = 'Level ' + l.level;
  $('#lv-pts').textContent = lvNum(l.into) + ' / ' + lvNum(l.need);
  $('#lv-fill').style.width = Math.max(2, Math.min(100, (l.into / l.need) * 100)) + '%';
  $('#lv-sub').textContent = l.today ? '+' + lvNum(l.today) + ' heute · ' + lvLine(l) : lvLine(l);
  // Unterseite
  const big = $('#lv-big'); if (!big) return;
  big.textContent = 'Level ' + l.level;
  $('#lv-big-fill').style.width = Math.max(2, Math.min(100, (l.into / l.need) * 100)) + '%';
  $('#lv-big-sub').textContent = lvNum(l.into) + ' von ' + lvNum(l.need) + ' Punkten · ' + lvLine(l) + ' · insgesamt ' + lvNum(l.xp);
  $('#lv-start').textContent = 'Punkte zählen seit ' + dayLabel(l.start) + '. Alle haben bei Level 1 angefangen.';
  const ul = $('#lv-list'); ul.replaceChildren();
  ul.hidden = !l.recent.length;
  $('#lv-empty').hidden = !!l.recent.length;
  for (const e of l.recent) {
    const li = el('li');
    const left = el('div', 'lv-what');
    if (e.kind === 'woche') left.append(el('b', null, 'Wochenbonus'), el('span', 'note', dayLabel(e.date) + ' · ' + e.note));
    else if (e.kind === 'abzeichen') left.append(el('b', null, e.name), el('span', 'note', dayLabel(e.date) + ' · Abzeichen'));
    else {
      const sp = sportOf(e.type || '');
      const bits = [dayLabel(e.date), lvNum(e.min) + ' Min.' + (e.hard ? ' · hart ×1,5' : '')];
      if (e.note) bits.push(e.note);
      left.append(el('b', null, e.name || sp.label), el('span', 'note', bits.join(' · ')));
    }
    li.append(left, el('span', 'lv-plus', '+' + lvNum(e.pts)));
    ul.append(li);
  }
  renderBadges(l.badges || []);
}

// ---------- Abzeichen (#192, Schritt 2) ----------
// Raster mit allen Abzeichen: Ring in der besten Stufe (Bronze/Silber/Gold, grau = noch keine), Fortschritt bis zur nächsten Stufe.
// Monats-Abzeichen zeigen diesen Monat und die Monate, in denen es schon eine Stufe gab.
const TIERS = ['Bronze', 'Silber', 'Gold'];
const monthName = (m) => MONTHS[Number(m.slice(5, 7)) - 1] + (m.slice(0, 4) !== S.today.slice(0, 4) ? ' ' + m.slice(2, 4) : '');
function badgeProgress(b) {
  const show = b.monthly ? b.nowTier : b.tier, cur = b.value;
  if (b.key === 'lang') return show >= 3 ? 'Alle Stufen erreicht' : 'Nächste Stufe: ' + ['Lauf ab 1:30 h oder Rad ab 3 h', 'Lauf ab 2:00 h oder Rad ab 4 h', 'Lauf ab 3:00 h oder Rad ab 6 h'][show];
  if (show >= 3) return (b.monthly ? monthName(S.today.slice(0, 7)) + ': ' : '') + lvNum(cur) + (b.unit ? ' ' + b.unit : '') + ' · Gold';
  return (b.monthly ? monthName(S.today.slice(0, 7)) + ': ' : '') + lvNum(cur) + ' / ' + lvNum(b.tiers[show]) + (b.unit ? ' ' + b.unit : '') + ' bis ' + TIERS[show];
}
function renderBadges(list) {
  const box = $('#lv-badges'); if (!box) return;
  box.replaceChildren();
  const got = list.filter((b) => b.tier > 0).length;
  $('#lv-badges-h').textContent = 'Abzeichen (' + got + ' von ' + list.length + ')';
  for (const b of list) {
    const show = b.monthly ? b.nowTier : b.tier;
    const t = el('div', 'bd-tile');
    const medal = el('span', 'bd-medal t' + b.tier); medal.innerHTML = icon(b.icon);
    const txt = el('div', 'bd-txt');
    txt.append(el('b', null, b.title), el('span', 'bd-tier', b.tier ? TIERS[b.tier - 1] + (b.monthly ? ' (beste Stufe)' : '') : 'Noch keine Stufe'));
    const next = show >= 3 ? 1 : b.key === 'lang' ? show / 3 : Math.min(1, b.value / b.tiers[show]);
    const bar = el('span', 'lv-bar'); const fill = el('span'); fill.style.width = Math.max(2, next * 100) + '%'; bar.append(fill);
    txt.append(bar, el('span', 'note', badgeProgress(b)));
    if (b.text) txt.append(el('span', 'note', b.text));
    if (b.monthly && b.months && b.months.length) txt.append(el('span', 'note', 'Bisher: ' + b.months.slice(0, 4).map((x) => monthName(x.m) + ' ' + TIERS[x.tier - 1]).join(' · ')));
    txt.append(el('span', 'note bd-desc', b.key === 'lang' ? b.desc : b.desc + ' · ' + b.tiers.join(' / ') + (b.unit ? ' ' + b.unit : '')));
    t.append(medal, txt);
    box.append(t);
  }
}

function renderLevelToday() {
  const b = $('#lv-today'); if (!b) return;
  const l = S.level;
  b.hidden = !(l && l.today > 0);
  if (b.hidden) return;
  b.replaceChildren(el('b', null, '+' + lvNum(l.today) + ' Punkte'), document.createTextNode(' · Level ' + l.level + ' · ' + lvLine(l)));
}

function initLevel() {
  const b = $('#lv-today');
  if (b) b.addEventListener('click', () => { S.kontoView = 'level'; showTab('konto'); showKontoView('level'); });
}
