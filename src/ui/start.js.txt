$('#key-save').addEventListener('click', async () => {
  const input = $('#key-input'), msg = $('#key-msg'), key = input.value.trim();
  if (!key) { msg.textContent = 'Bitte zuerst den Schlüssel einfügen.'; return; }
  if (!$('#key-term').value && !$('#key-expires').value) { msg.textContent = 'Bitte noch die Laufzeit wählen, die du bei Garmin eingestellt hast (oder „Weiß ich nicht“).'; return; }
  msg.textContent = 'Prüfe Schlüssel bei Garmin …';
  try {
    const r = await api('/api/key', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ key, expires: $('#key-expires').value || null }) });
    input.value = '';
    if (r.ok) { msg.textContent = 'Gespeichert. Daten werden geladen …'; await syncLoop(true); msg.textContent = 'Schlüssel gespeichert.'; }
    else msg.textContent = r.error || 'Speichern fehlgeschlagen.';
  } catch (e) { input.value = ''; msg.textContent = 'Fehler: ' + e.message; }
});

$('#zone-save').addEventListener('click', async () => {
  const msg = $('#zone-msg');
  const v = [...document.querySelectorAll('#zone-form input')].map((i) => Number(i.value));
  const zones = [[v[0], v[1]], [v[2], v[3]], [v[4], v[5]]];
  const ok = v.every((x) => Number.isInteger(x) && x >= 60 && x <= 230) && zones.every((z) => z[0] <= z[1]) && zones[0][1] < zones[1][0] && zones[1][1] < zones[2][0];
  if (!ok) { msg.textContent = 'Bitte ganze Pulswerte zwischen 60 und 230 eintragen, aufsteigend und ohne Überschneidung.'; return; }
  msg.textContent = 'Speichere …';
  try {
    const r = await api('/api/zones', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ zones }) });
    if (r.ok) { S.zoneCfg = r.zones; renderZoneForm(true); msg.textContent = 'Gespeichert. Zonen-Diagramm und „Pace je Zone“ rechnen jetzt mit diesen Werten.'; loadData().catch(() => {}); }
    else msg.textContent = r.error || 'Speichern fehlgeschlagen.';
  } catch (e) { msg.textContent = 'Fehler: ' + e.message; }
});

});
