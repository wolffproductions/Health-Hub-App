// Status der Garmin-Verbindung für die Übersicht: Ampel-Punkt, kurzer Text, zweite Zeile
function garminSourceStatus() {
  const st = S.status;
  if (!st) return { light: '', text: 'Lädt …', sub: '' };
  if (st.key === 'none') return { light: 'r', text: 'Nicht verbunden', sub: 'Schlüssel eintragen' };
  const exp = keyExpiry(st);
  const last = st.lastRecent ? 'Aktualisiert ' + relTime(st.lastRecent) : 'Noch nicht aktualisiert';
  const hist = st.complete ? '' : ' · Historie wird geladen';
  if (st.lastError && /abgelehnt|401|403/.test(st.lastError)) return { light: 'r', text: 'Schlüssel abgelehnt', sub: 'Neuen Schlüssel eintragen' };
  if (exp && exp.left < 0) return { light: 'r', text: 'Schlüssel abgelaufen', sub: 'Seit ' + exp.label };
  if (exp && exp.left <= KEY_WARN_DAYS) return { light: 'y', text: exp.left === 0 ? 'Läuft heute ab' : 'Läuft in ' + exp.left + (exp.left === 1 ? ' Tag' : ' Tagen') + ' ab', sub: last + hist };
  if (st.lastError) return { light: 'y', text: 'Verbunden, letzter Abruf mit Fehler', sub: last };
  return { light: 'g', text: 'Verbunden', sub: last + hist + (exp ? ' · Schlüssel bis ' + exp.label : '') };
}
function relTime(iso) {
  const t = new Date(iso), d = t.toLocaleDateString('en-CA'), today = new Date().toLocaleDateString('en-CA');
  const time = t.toLocaleTimeString('de-DE', { hour: '2-digit', minute: '2-digit' });
  return d === today ? 'heute ' + time : d === addDays(today, -1) ? 'gestern ' + time : dayLabel(d) + ' ' + time;
}

function renderSources() {
  const ul = $('#sources'); ul.replaceChildren();
  for (const src of SOURCES) {
    const st = src.status();
    const li = el('li');
    const row = el(src.detail ? 'button' : 'div', 'srcrow' + (src.detail ? '' : ' off'));
    if (src.detail) { row.type = 'button'; row.addEventListener('click', () => showKontoView(src.detail)); }
    const ic = el('span', 'srcicon'); ic.innerHTML = icon(SOURCE_ICONS[src.icon]); ic.style.background = css(src.color);
    const txt = el('span', 'srctext'); txt.append(el('span', 'srcname', src.name)); if (st.sub) txt.append(el('span', 'srcsub', st.sub));
    const right = el('span', 'srcstate');
    if (st.light) right.append(el('i', 'light ' + st.light));
    right.append(el('span', null, st.text));
    row.append(ic, txt, right);
    if (src.detail) { const ch = el('span', 'chev'); ch.innerHTML = icon('weiter'); row.append(ch); }
    li.append(row); ul.append(li);
  }
}

// Welche Quelle zählt nachts (Schlaf, HRV, Ruhepuls, SpO2), wenn mehrere Geräte messen; auf WHOOP- und Google-Seite gleich
const NIGHT_NAMES = { whoop: 'WHOOP', garmin: 'Garmin', google: 'Google Health / Fitbit' };
async function saveNightSource(night, msg) {
  try {
    await api('/api/sources', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ night }) });
    $(msg).textContent = 'Gespeichert. Nachtwerte kommen jetzt von ' + (NIGHT_NAMES[night] || night) + '.';
    await loadData();
  } catch { $(msg).textContent = 'Speichern fehlgeschlagen.'; }
}

  if (g.expired) return { light: 'r', text: 'Verbindung abgelaufen', sub: 'Neu verbinden' };
  if (!g.connected) return { light: 'r', text: 'Nicht verbunden', sub: 'Mit Google verbinden' };
  const last = g.lastSleep ? 'Letzte Nacht ' + relTime(g.lastSleep) : 'Noch keine Nacht';
  const left = googleDaysLeft(g);
  // Kurz halten, sonst schneidet die Liste den Namen ab
  if (left != null && left <= 2) return { light: 'y', text: left <= 0 ? 'Läuft heute ab' : 'Noch ' + left + (left === 1 ? ' Tag' : ' Tage'), sub: 'Neu verbinden' };
  if (g.paused) return { light: 'y', text: 'Pause (Google-Limit)', sub: last };
  if (g.lastError) return { light: 'y', text: 'Verbunden, letzter Abruf mit Fehler', sub: last };
  return { light: 'g', text: 'Verbunden', sub: last + (g.complete ? '' : ' · Historie wird geladen') };
}

function bytesToB64u(buf) { return btoa(String.fromCharCode(...new Uint8Array(buf))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, ''); }
// Gerätename für die Liste, aus dem Browser abgeleitet
function deviceName() {
  const ua = navigator.userAgent;
  if (/iPhone/.test(ua)) return 'iPhone';
  if (/iPad/.test(ua) || (/Macintosh/.test(ua) && navigator.maxTouchPoints > 1)) return 'iPad';
  if (/Macintosh/.test(ua)) return 'Mac';
  if (/Android/.test(ua)) return 'Android';
  if (/Windows/.test(ua)) return 'Windows';
  return 'Gerät';
}
async function loadPasskeys() {
  let keys;
  try { keys = await api('/api/passkeys'); } catch { $('#pk-state').textContent = 'Passkeys konnten nicht geladen werden.'; return; }
  if (!Array.isArray(keys)) keys = [];
  const ul = $('#pk-list'); ul.replaceChildren();
  ul.hidden = !keys.length;
  const can = !!(window.PublicKeyCredential && navigator.credentials);
  $('#pk-add').disabled = !can;
  $('#pk-state').textContent = !can ? 'Dieser Browser unterstützt keine Passkeys.'
    : keys.length ? `${keys.length} ${keys.length === 1 ? 'Passkey' : 'Passkeys'} eingetragen. Auf der Login-Seite erscheint „Mit Passkey anmelden“.`
    : 'Noch kein Passkey eingetragen.';
  $('#m-anmeldung').textContent = keys.length ? 'Passkey und Passwort' : 'Passwort';
  for (const k of keys) {
    const li = el('li');
    const left = el('div');
    left.append(el('div', null, k.name));
    left.append(el('div', 'meta', 'Angelegt ' + new Date(k.created).toLocaleDateString('de-DE') + ' · ' + (k.used ? 'zuletzt ' + relTime(k.used) : 'noch nicht benutzt')));
    const del = el('button', 'btn small danger', 'Entfernen'); del.type = 'button';
    del.addEventListener('click', async () => {
      if (!confirm(`Passkey „${k.name}“ entfernen? Auf dem Gerät bleibt er im Schlüsselbund, hier funktioniert er dann nicht mehr.`)) return;
      del.disabled = true;
      const r = await api('/api/passkeys?id=' + encodeURIComponent(k.id), { method: 'DELETE' }).catch(() => ({}));
      $('#pk-msg').textContent = r.ok ? 'Passkey entfernt.' : (r.error || 'Entfernen fehlgeschlagen.');
      loadPasskeys();
    });
    li.append(left, del); ul.append(li);
  }
}
function initPasskeys() {
  $('#pk-add').addEventListener('click', async () => {
    const b = $('#pk-add'); b.disabled = true; $('#pk-msg').textContent = '';
    try {
      const o = await api('/api/passkeys/options', { method: 'POST' });
      if (o.error) { $('#pk-msg').textContent = o.error; return; }
      const cred = await navigator.credentials.create({ publicKey: {
        ...o,
        challenge: b64uToBytes(o.challenge),
        user: { ...o.user, id: b64uToBytes(o.user.id) },
        excludeCredentials: o.excludeCredentials.map((c) => ({ ...c, id: b64uToBytes(c.id) })),
      } });
      const r = await api('/api/passkeys', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({
        challenge: o.challenge, id: cred.id, name: deviceName(),
        attestationObject: bytesToB64u(cred.response.attestationObject), clientDataJSON: bytesToB64u(cred.response.clientDataJSON) }) });
      $('#pk-msg').textContent = r.ok ? 'Passkey angelegt. Ab jetzt kannst du dich mit Face ID anmelden.' : (r.error || 'Anlegen fehlgeschlagen.');
    } catch (e) {
      $('#pk-msg').textContent = e && e.name === 'InvalidStateError' ? 'Auf diesem Gerät ist schon ein Passkey für die App eingetragen.'
        : e && e.name === 'NotAllowedError' ? 'Abgebrochen.' : 'Anlegen fehlgeschlagen.';
    } finally { b.disabled = false; loadPasskeys(); }
  });
  loadPasskeys();
}

// Großansicht eines Profilbilds (eigenes oder eines Freundes, #87); liefert das Bild, damit eine schärfere Fassung nachkommen kann
function zoomAvatar(src, name) {
  if (!src || document.querySelector('.av-view')) return null;
  const v = el('div', 'av-view'); v.setAttribute('role', 'dialog'); v.setAttribute('aria-modal', 'true'); v.setAttribute('aria-label', 'Profilbild');
  const im = el('img'); im.src = src; im.alt = name ? 'Profilbild von ' + name : 'Profilbild';
  im.draggable = false;
  // Zoomen wie bei Instagram: zwei Finger ziehen den GANZEN Kreis samt Bild größer und schieben ihn mit,
  // beim Loslassen gleitet er weich zurück. Am Rechner Mausrad (springt nach kurzer Pause zurück).
  const frame = el('div', 'av-frame'); frame.append(im); v.append(frame);
  if (name) v.append(el('div', 'av-view-name', name));
  let s = 1, tx = 0, ty = 0, wheelT = 0;
  const pts = new Map(); let start = null;
  const apply = (anim) => {
    frame.style.animation = 'none';
    frame.style.transition = anim ? 'transform .32s cubic-bezier(.2,.8,.2,1)' : 'none';
    frame.style.transform = `translate(${tx}px, ${ty}px) scale(${s})`;
  };
  const back = () => { s = 1; tx = 0; ty = 0; apply(true); };
  const mid = (p) => [(p[0][0] + p[1][0]) / 2, (p[0][1] + p[1][1]) / 2];
  frame.addEventListener('pointerdown', (e) => {
    frame.setPointerCapture(e.pointerId); pts.set(e.pointerId, [e.clientX, e.clientY]);
    const p = [...pts.values()];
    if (p.length === 2) start = { d: Math.hypot(p[0][0] - p[1][0], p[0][1] - p[1][1]) || 1, s, tx, ty, m: mid(p) };
  });
  frame.addEventListener('pointermove', (e) => {
    if (!pts.has(e.pointerId)) return;
    pts.set(e.pointerId, [e.clientX, e.clientY]);
    const p = [...pts.values()];
    if (p.length !== 2 || !start) return;
    const d = Math.hypot(p[0][0] - p[1][0], p[0][1] - p[1][1]), m = mid(p);
    s = Math.max(1, Math.min(3, start.s * (d / start.d)));
    tx = start.tx + m[0] - start.m[0]; ty = start.ty + m[1] - start.m[1];
    apply(false);
  });
  const up = (e) => { pts.delete(e.pointerId); if (pts.size < 2) { start = null; back(); } };
  frame.addEventListener('pointerup', up); frame.addEventListener('pointercancel', up);
  frame.addEventListener('wheel', (e) => {
    e.preventDefault(); s = Math.max(1, Math.min(3, s * Math.exp(-e.deltaY / 300))); apply(false);
    clearTimeout(wheelT); wheelT = setTimeout(back, 350);
  }, { passive: false });
  // Tippen auf das Bild selbst schließt nicht (nur daneben)
  frame.addEventListener('click', (e) => e.stopPropagation());
  const close = () => { v.classList.add('zu'); document.removeEventListener('keydown', esc); setTimeout(() => v.remove(), 180); };
  const esc = (e) => { if (e.key === 'Escape') close(); };
  // Tippen neben den Kreis schließt
  v.addEventListener('click', close); document.addEventListener('keydown', esc);
  document.body.append(v);
  return im;
}

// ---------- Profilbild zuschneiden ----------
// Nach der Auswahl erscheint das Foto mit runder Schablone: ein Finger verschiebt, zwei Finger zoomen (am Rechner das Mausrad).
// „Übernehmen“ schneidet genau den Kreis-Ausschnitt heraus (quadratisch 256 × 256 px, JPEG), „Abbrechen“ verwirft.
function cropImage(file) {
  return new Promise((resolve, reject) => {
    const url = URL.createObjectURL(file), im = new Image();
    im.onerror = () => { URL.revokeObjectURL(url); reject(new Error('Bild konnte nicht gelesen werden')); };
    im.onload = () => {
      const W = im.naturalWidth, H = im.naturalHeight;
      const ov = el('div', 'crop'); ov.setAttribute('role', 'dialog'); ov.setAttribute('aria-modal', 'true'); ov.setAttribute('aria-label', 'Bild zuschneiden');
      const stage = el('div', 'crop-stage');
      const pic = el('img', 'crop-img'); pic.src = url; pic.alt = ''; pic.draggable = false;
      const mask = el('div', 'crop-mask'); mask.setAttribute('aria-hidden', 'true');
      stage.append(pic, mask);
      const hint = el('p', 'crop-hint', 'Verschieben und mit zwei Fingern zoomen');
      const row = el('div', 'crop-row');
      const cancel = el('button', 'btn ghost', 'Abbrechen'); cancel.type = 'button';
      const ok = el('button', 'btn', 'Übernehmen'); ok.type = 'button';
      row.append(cancel, ok);
      ov.append(el('div', 'crop-title', 'Bild zuschneiden'), stage, hint, row);
      document.body.append(ov); document.body.classList.add('crop-open');

      const V = stage.clientWidth; // Kantenlänge des Ausschnitts (Kreis passt genau hinein)
      const base = V / Math.min(W, H); // Grundmaßstab: kurze Seite füllt den Kreis
      let z = 1, x = (V - W * base) / 2, y = (V - H * base) / 2;
      const clamp = () => {
        const w = W * base * z, h = H * base * z;
        x = Math.min(0, Math.max(V - w, x)); y = Math.min(0, Math.max(V - h, y));
      };
      const draw = () => { clamp(); pic.style.width = W * base * z + 'px'; pic.style.height = H * base * z + 'px'; pic.style.transform = `translate(${x}px, ${y}px)`; };
      // Zoom um einen Punkt (cx, cy im Ausschnitt), damit die Stelle unter den Fingern stehen bleibt
      const zoomTo = (nz, cx = V / 2, cy = V / 2) => {
        nz = Math.min(4, Math.max(1, nz));
        x = cx - (cx - x) * (nz / z); y = cy - (cy - y) * (nz / z); z = nz;
        draw();
      };
      draw();

      const pts = new Map();
      let last = null; // { x, y, d } beim letzten Bewegen
      const rel = (e) => { const r = stage.getBoundingClientRect(); return { x: e.clientX - r.left, y: e.clientY - r.top }; };
      const state = () => {
        const p = [...pts.values()];
        if (p.length === 1) return { x: p[0].x, y: p[0].y, d: null };
        const [a, b] = p;
        return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2, d: Math.hypot(a.x - b.x, a.y - b.y) };
      };
      stage.addEventListener('pointerdown', (e) => { stage.setPointerCapture(e.pointerId); pts.set(e.pointerId, rel(e)); last = state(); });
      stage.addEventListener('pointermove', (e) => {
        if (!pts.has(e.pointerId)) return;
        pts.set(e.pointerId, rel(e));
        const s = state();
        if (last) {
          x += s.x - last.x; y += s.y - last.y;
          if (s.d && last.d) zoomTo(z * (s.d / last.d), s.x, s.y); else draw();
        }
        last = s;
      });
      const up = (e) => { pts.delete(e.pointerId); last = pts.size ? state() : null; };
      stage.addEventListener('pointerup', up); stage.addEventListener('pointercancel', up);
      stage.addEventListener('wheel', (e) => { e.preventDefault(); const p = rel(e); zoomTo(z * (e.deltaY < 0 ? 1.08 : 1 / 1.08), p.x, p.y); }, { passive: false });

      const done = (val) => { ov.remove(); document.body.classList.remove('crop-open'); URL.revokeObjectURL(url); resolve(val); };
      cancel.addEventListener('click', () => done(null));
      // Zwei Größen: klein (256 px) für oben rechts, Listen und Freunde, groß (bis 1024 px, nie größer als das Original)
      // für die Großansicht, damit sie auch auf dem iPhone-Bildschirm scharf ist
      ok.addEventListener('click', () => {
        const s = base * z, src = V / s;
        const shot = (px, q) => { const c = document.createElement('canvas'); c.width = c.height = px; const g = c.getContext('2d'); g.imageSmoothingQuality = 'high'; g.drawImage(im, -x / s, -y / s, src, src, 0, 0, px, px); return c.toDataURL('image/jpeg', q); };
        done({ img: shot(256, 0.85), big: shot(Math.max(256, Math.min(1024, Math.round(src))), 0.85) });
      });
    };
    im.src = url;
  });
}

function fbView() { const t = FB.from || S.lastTab || 'heute'; return (TAB_TITLES[t] || t) + (t === 'training' || t === 'leistung' || t === 'schlaf' || t === 'freunde' ? ' → ' + { training: 'Training', leistung: 'Leistung', schlaf: 'Schlaf', freunde: 'Freunde' }[t] : ''); }
function fbDevice() {
  const ua = navigator.userAgent;
  const os = /iPhone|iPad/.test(ua) ? (ua.match(/OS (\d+)_(\d+)/) ? 'iOS ' + RegExp.$1 + '.' + RegExp.$2 : 'iOS') : /Android/.test(ua) ? 'Android' : /Mac/.test(ua) ? 'Mac' : /Windows/.test(ua) ? 'Windows' : 'Gerät';
  const br = /CriOS|Chrome/.test(ua) ? 'Chrome' : /FxiOS|Firefox/.test(ua) ? 'Firefox' : /Safari/.test(ua) ? 'Safari' : 'Browser';
  const app = window.matchMedia && matchMedia('(display-mode: standalone)').matches ? ', Homescreen-App' : '';
  return os + ', ' + br + app + ', ' + window.innerWidth + ' px';
}
// Bild verkleinern: längste Seite höchstens 1280 px, JPEG; bei Bedarf stärker komprimieren (Ziel unter 1,5 MB)
function fbShrink(file) {
  return new Promise((resolve, reject) => {
    const url = URL.createObjectURL(file), im = new Image();
    im.onerror = () => { URL.revokeObjectURL(url); reject(new Error('Bild konnte nicht gelesen werden')); };
    im.onload = () => {
      const f = Math.min(1, 1280 / Math.max(im.naturalWidth, im.naturalHeight));
      const c = document.createElement('canvas'); c.width = Math.round(im.naturalWidth * f); c.height = Math.round(im.naturalHeight * f);
      c.getContext('2d').drawImage(im, 0, 0, c.width, c.height); URL.revokeObjectURL(url);
      let q = 0.82, out = c.toDataURL('image/jpeg', q);
      while (out.length > 2_000_000 && q > 0.4) { q -= 0.12; out = c.toDataURL('image/jpeg', q); }
      resolve(out);
    };
    im.src = url;
  });
}
