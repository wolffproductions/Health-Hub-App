// Status der Garmin-Verbindung für die Übersicht: Ampel-Punkt, kurzer Text, zweite Zeile
function garminSourceStatus() {
  const st = S.status;
  if (!st) return { light: '', text: 'Lädt …', sub: '' };
  if (st.key === 'none') return { light: 'r', text: 'Nicht verbunden', sub: 'Schlüssel eintragen' };
  const exp = keyExpiry(st);
  const last = st.lastRecent ? 'Aktualisiert ' + relTime(st.lastRecent) : 'Noch nicht aktualisiert';
  const hist = st.complete ? '' : ' · Historie wird geladen';
  if (st.lastError && /abgelehnt|401|403/.test(st.lastError)) return { light: 'r', text: 'Schlüssel abgelehnt', sub: 'Neuen Schlüssel eintragen' };
  if (exp && exp.left < 0) return { light: 'r', text: 'Schlüssel abgelaufen', sub: 'Seit ' + exp.label };
  if (exp && exp.left <= KEY_WARN_DAYS) return { light: 'y', text: exp.left === 0 ? 'Läuft heute ab' : 'Läuft in ' + exp.left + (exp.left === 1 ? ' Tag' : ' Tagen') + ' ab', sub: last + hist };
  if (st.lastError) return { light: 'y', text: 'Verbunden, letzter Abruf mit Fehler', sub: last };
  return { light: 'g', text: 'Verbunden', sub: last + hist + (exp ? ' · Schlüssel bis ' + exp.label : '') };
}
function relTime(iso) {
  const t = new Date(iso), d = t.toLocaleDateString('en-CA'), today = new Date().toLocaleDateString('en-CA');
  const time = t.toLocaleTimeString('de-DE', { hour: '2-digit', minute: '2-digit' });
  return d === today ? 'heute ' + time : d === addDays(today, -1) ? 'gestern ' + time : dayLabel(d) + ' ' + time;
}

function renderSources() {
  const ul = $('#sources'); ul.replaceChildren();
  for (const src of SOURCES) {
    const st = src.status();
    const li = el('li');
    const row = el(src.detail ? 'button' : 'div', 'srcrow' + (src.detail ? '' : ' off'));
    if (src.detail) { row.type = 'button'; row.addEventListener('click', () => showKontoView(src.detail)); }
    const ic = el('span', 'srcicon'); ic.innerHTML = icon(SOURCE_ICONS[src.icon]); ic.style.background = css(src.color);
    const txt = el('span', 'srctext'); txt.append(el('span', 'srcname', src.name)); if (st.sub) txt.append(el('span', 'srcsub', st.sub));
    const right = el('span', 'srcstate');
    if (st.light) right.append(el('i', 'light ' + st.light));
    right.append(el('span', null, st.text));
    row.append(ic, txt, right);
    if (src.detail) { const ch = el('span', 'chev'); ch.innerHTML = icon('weiter'); row.append(ch); }
    li.append(row); ul.append(li);
  }
}

// ---------- Morgen-Push (#13) ----------
// Web Push an die Homescreen-App. Auf dem iPhone nur, wenn die App über „Zum Home-Bildschirm“ geöffnet wurde (ab iOS 16.4).
const PUSH = { info: null };
const standalone = () => window.matchMedia('(display-mode: standalone)').matches || navigator.standalone === true;
const pushSupported = () => 'serviceWorker' in navigator && 'PushManager' in window && 'Notification' in window;
function b64uToBytes(s) { const b = s.replace(/-/g, '+').replace(/_/g, '/'); return Uint8Array.from(atob(b + '==='.slice((b.length + 3) % 4)), (c) => c.charCodeAt(0)); }
async function currentSub() { try { const reg = await navigator.serviceWorker.ready; return await reg.pushManager.getSubscription(); } catch { return null; } }

async function renderPush() {
  const sel = $('#push-time');
  if (!sel.options.length) for (let m = 5 * 60; m <= 12 * 60; m += 30) { const t = String(Math.floor(m / 60)).padStart(2, '0') + ':' + String(m % 60).padStart(2, '0'); sel.append(new Option(t + ' Uhr', t)); }
  try { PUSH.info = await api('/api/push'); } catch { PUSH.info = null; }
  const i = PUSH.info || { on: false, time: '09:00', devices: 0 };
  sel.value = i.time;
  const sub = pushSupported() ? await currentSub() : null;
  const on = !!(i.on && sub);
  let state;
  if (!pushSupported()) state = standalone() ? 'Dieses Gerät unterstützt keine Push-Meldungen (auf dem iPhone ab iOS 16.4).' : 'Auf dem iPhone: in Safari „Teilen → Zum Home-Bildschirm“ wählen und die App von dort öffnen. Erst dann sind Push-Meldungen möglich.';
  else if (Notification.permission === 'denied') state = 'Mitteilungen sind für die App ausgeschaltet. Erlaube sie in den iPhone-Einstellungen unter „Mitteilungen → Health“.';
  else if (on) state = `Aktiv: jeden Morgen um ${i.time} Uhr${i.devices > 1 ? ' an ' + i.devices + ' Geräte' : ''}.` + (i.lastSent ? ' Zuletzt gesendet: ' + dayLabel(i.lastSent) + '.' : '') + (i.lastResult && !/^Gesendet/.test(i.lastResult) ? ' ' + i.lastResult + '.' : '');
  else state = i.devices ? 'Auf einem anderen Gerät aktiv. Hier noch nicht eingeschaltet.' : 'Aus.';
  $('#push-state').textContent = state;
  $('#push-on').textContent = on ? 'Ausschalten' : 'Push aktivieren';
  $('#push-on').classList.toggle('ghost', on);
  $('#push-on').hidden = !pushSupported() || Notification.permission === 'denied';
  $('#push-test').hidden = !on;
}

function initPush() {
  if ('serviceWorker' in navigator) navigator.serviceWorker.register('/sw.js').catch(() => {});
  $('#push-on').addEventListener('click', async () => {
    const b = $('#push-on'), msg = $('#push-msg'); b.disabled = true; msg.textContent = '';
    try {
      const sub = await currentSub();
      if (sub && PUSH.info && PUSH.info.on) {
        await api('/api/push/unsubscribe', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ endpoint: sub.endpoint }) });
        await sub.unsubscribe().catch(() => {});
        msg.textContent = 'Ausgeschaltet.';
      } else {
        // Erlaubnis muss direkt aus dem Antippen heraus angefragt werden (iOS)
        const perm = await Notification.requestPermission();
        if (perm !== 'granted') { msg.textContent = 'Ohne Erlaubnis keine Meldungen.'; return; }
        const reg = await navigator.serviceWorker.ready;
        const key = (PUSH.info && PUSH.info.key) || (await api('/api/push')).key;
        const s = sub || await reg.pushManager.subscribe({ userVisibleOnly: true, applicationServerKey: b64uToBytes(key) });
        const r = await api('/api/push/subscribe', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ sub: s.toJSON(), time: $('#push-time').value }) });
        msg.textContent = r.ok ? 'Eingeschaltet. Tipp: „Test senden“ zeigt dir die Meldung sofort.' : (r.error || 'Fehlgeschlagen.');
      }
    } catch (e) { msg.textContent = 'Fehler: ' + e.message; }
    finally { b.disabled = false; renderPush(); }
  });
  $('#push-time').addEventListener('change', async () => {
    await api('/api/push/settings', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ time: $('#push-time').value }) }).catch(() => {});
    $('#push-msg').textContent = 'Uhrzeit gespeichert.'; renderPush();
  });
  $('#push-test').addEventListener('click', async () => {
    const b = $('#push-test'); b.disabled = true; $('#push-msg').textContent = 'Wird gesendet …';
    try { const r = await api('/api/push/test', { method: 'POST' }); $('#push-msg').textContent = r.result || 'Gesendet.'; }
    catch (e) { $('#push-msg').textContent = 'Fehler: ' + e.message; } finally { b.disabled = false; }
  });
}

// ---------- Claude-Zugang (#5) ----------
async function renderClaude() {
  let i; try { i = await api('/api/mcp'); } catch { return; }
  $('#mcp-url').value = i.url;
  $('#mcp-mood').checked = !!i.mood;
  const ul = $('#mcp-clients'); ul.replaceChildren();
  ul.hidden = !i.clients.length;
  for (const c of i.clients) {
    const li = el('li'), txt = el('span', 'mcp-txt');
    txt.append(el('span', null, c.name), el('span', 'mcp-sub', c.lastUsed ? 'Zuletzt ' + relTime(c.lastUsed) : 'Verbunden ' + relTime(c.created)));
    const b = el('button', 'btn small ghost', 'Entfernen'); b.type = 'button';
    b.addEventListener('click', async () => {
      if (!confirm(c.name + ' den Zugriff entziehen?')) return;
      await fetch('/api/mcp?client=' + encodeURIComponent(c.id), { method: 'DELETE', credentials: 'same-origin' });
      $('#mcp-msg').textContent = 'Zugriff entzogen.'; renderClaude();
    });
    li.append(txt, b); ul.append(li);
  }
}

  $('#mcp-mood').addEventListener('change', async (e) => {
    try {
      await api('/api/mcp', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ mood: e.target.checked }) });
      $('#mcp-msg').textContent = e.target.checked ? 'Claude darf jetzt deine Stimmung lesen.' : 'Stimmung ist für Claude wieder gesperrt.';
    } catch { $('#mcp-msg').textContent = 'Speichern fehlgeschlagen.'; }
  });
}

function bytesToB64u(buf) { return btoa(String.fromCharCode(...new Uint8Array(buf))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, ''); }
// Gerätename für die Liste, aus dem Browser abgeleitet
function deviceName() {
  const ua = navigator.userAgent;
  if (/iPhone/.test(ua)) return 'iPhone';
  if (/iPad/.test(ua) || (/Macintosh/.test(ua) && navigator.maxTouchPoints > 1)) return 'iPad';
  if (/Macintosh/.test(ua)) return 'Mac';
  if (/Android/.test(ua)) return 'Android';
  if (/Windows/.test(ua)) return 'Windows';
  return 'Gerät';
}
async function loadPasskeys() {
  let keys;
  try { keys = await api('/api/passkeys'); } catch { $('#pk-state').textContent = 'Passkeys konnten nicht geladen werden.'; return; }
  if (!Array.isArray(keys)) keys = [];
  const ul = $('#pk-list'); ul.replaceChildren();
  ul.hidden = !keys.length;
  const can = !!(window.PublicKeyCredential && navigator.credentials);
  $('#pk-add').disabled = !can;
  $('#pk-state').textContent = !can ? 'Dieser Browser unterstützt keine Passkeys.'
    : keys.length ? `${keys.length} ${keys.length === 1 ? 'Passkey' : 'Passkeys'} eingetragen. Auf der Login-Seite erscheint „Mit Passkey anmelden“.`
    : 'Noch kein Passkey eingetragen.';
  for (const k of keys) {
    const li = el('li');
    const left = el('div');
    left.append(el('div', null, k.name));
    left.append(el('div', 'meta', 'Angelegt ' + new Date(k.created).toLocaleDateString('de-DE') + ' · ' + (k.used ? 'zuletzt ' + relTime(k.used) : 'noch nicht benutzt')));
    const del = el('button', 'btn small danger', 'Entfernen'); del.type = 'button';
    del.addEventListener('click', async () => {
      if (!confirm(`Passkey „${k.name}“ entfernen? Auf dem Gerät bleibt er im Schlüsselbund, hier funktioniert er dann nicht mehr.`)) return;
      del.disabled = true;
      const r = await api('/api/passkeys?id=' + encodeURIComponent(k.id), { method: 'DELETE' }).catch(() => ({}));
      $('#pk-msg').textContent = r.ok ? 'Passkey entfernt.' : (r.error || 'Entfernen fehlgeschlagen.');
      loadPasskeys();
    });
    li.append(left, del); ul.append(li);
  }
}
function initPasskeys() {
  $('#pk-add').addEventListener('click', async () => {
    const b = $('#pk-add'); b.disabled = true; $('#pk-msg').textContent = '';
    try {
      const o = await api('/api/passkeys/options', { method: 'POST' });
      if (o.error) { $('#pk-msg').textContent = o.error; return; }
      const cred = await navigator.credentials.create({ publicKey: {
        ...o,
        challenge: b64uToBytes(o.challenge),
        user: { ...o.user, id: b64uToBytes(o.user.id) },
        excludeCredentials: o.excludeCredentials.map((c) => ({ ...c, id: b64uToBytes(c.id) })),
      } });
      const r = await api('/api/passkeys', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({
        challenge: o.challenge, id: cred.id, name: deviceName(),
        attestationObject: bytesToB64u(cred.response.attestationObject), clientDataJSON: bytesToB64u(cred.response.clientDataJSON) }) });
      $('#pk-msg').textContent = r.ok ? 'Passkey angelegt. Ab jetzt kannst du dich mit Face ID anmelden.' : (r.error || 'Anlegen fehlgeschlagen.');
    } catch (e) {
      $('#pk-msg').textContent = e && e.name === 'InvalidStateError' ? 'Auf diesem Gerät ist schon ein Passkey für die App eingetragen.'
        : e && e.name === 'NotAllowedError' ? 'Abgebrochen.' : 'Anlegen fehlgeschlagen.';
    } finally { b.disabled = false; loadPasskeys(); }
  });
  loadPasskeys();
}
